add_namespace = italy.demands.savoy

add_namespace = italy.demands.istra

add_namespace = italy.demands.rejected

country_event = {
	id = italy.demands.savoy.1
	title = "Italy requests control of Savoy"
	picture = GFX_report_event_generic_sign_treaty3

	desc = {
		text = "Italy is requesting the reintegration of previously occupied italian territory of Savoy"
	}
	is_triggered_only = yes
	fire_only_once = no

	option = {
		name = "OK"

		ai_chance = {
			base = 10
		}

		if = {
			limit = {
				735 = {is_fully_controlled_by = GER}
			}
			
			ITA = {
				 transfer_state = 735
				 add_war_support = +0.01
				}
		}
	}

	option = {
		name = "No"
		ITA = { country_event = italy.demands.rejected.1 }
	}
}
