war_start_effects = {
	
	if = {
		limit = { 
			date > 1943.12.30
			NOT = { has_global_flag = mod_war_start_effects }
		}
		set_global_flag = mod_war_start_effects

		every_country = {
			limit = { is_in_faction_with = GER }
			add_to_war = {
				targeted_alliance = GER
				enemy = ENG
			}
		}	

		ENG = {
			every_country = {
				limit = { is_in_faction_with = ENG }
				add_to_war = {
					targeted_alliance = ENG
					enemy = GER
				}
			}
		}

		ENG = {
			every_country = {
				limit = { is_in_faction_with = ENG }
				add_to_war = {
					targeted_alliance = ENG
					enemy = JAP
				}
			}
		}
		
	}
}
