Resistance_start_effects = {
	
	if = {
		limit = { 
			date > 1944.1.3
			check_variable = { global.var_Resistance_start_effects = 0 }
		}


		if = {
			limit = {
				has_dlc = "La Resistance"
			}


				every_state = {
					limit = {
						controller = {has_war = yes}
						NOT = {is_core_of = controller}
						is_fully_controlled_by = controller

						OR = {
							controller = { is_in_faction_with = JAP }
							controller = { is_in_faction_with = GER }
						}
					}
					
					set_compliance = 24
					start_resistance = is_core_of
					set_resistance = 29
					
					force_enable_resistance = {
						clear = no
						occupier = controller
						occupied = is_core_of
					}

					#add_state_modifier = {
					#	modifier = { resistance_target = +0.25 }
					#}
				}

				104 = {
					set_compliance = 24
					start_resistance = yes
					set_resistance = 68
				}
		
				69 = {
					set_compliance = 64
					start_resistance = no
					set_resistance = 6
				}

				set_variable = { global.var_Resistance_start_effects = 1 }
		}


	}
}
