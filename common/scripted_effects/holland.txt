HOL_setup_resistance = {

		if = {
			limit = {
				date > 1943.12.30
				is_subject_of = GER
				NOT = { has_global_flag = HOL_setup_resistance }
			}
				set_global_flag = HOL_setup_resistance

			every_state = {
				limit = {
					is_core_of = THIS
					controller = {
						has_government = fascism
						tag = RKN
					}
				}

				force_enable_resistance = {
					occupier = THIS.controller
					occupied = THIS
				}

				start_resistance = THIS
				set_resistance = 11
				set_compliance = 61
				add_resistance_target = 21
			}
		}
