GER_scripted_events_start_effects = {
	
	if = {
		limit = { 
			date > 1943.12.30
			NOT = { has_global_flag = GER_scripted_events_start_effects }
		}
		set_global_flag = GER_scripted_events_start_effects
	}
}

GER_calculate_casualities_inflicted_on_SOV = {
	
	if = {
		limit = { 
			tag = GER
			has_war_with = SOV
		}

		SOV = { set_temp_variable = { SOV_casualties = casualties_k } }
		set_variable = { global.SOV_casualties_g = SOV_casualties }
	}
}

GER_operation_margarethe = {
	if = {
		limit = { 
			original_tag = GER
			date > 1944.3.18
			HUN = { is_ai = yes } 
			HUN = { is_in_faction_with = GER }

			has_war_with = SOV
			NOT = { has_global_flag = GER_operation_margarethe } 
		}

		GER = {
			country_event = {
				id = operation.margarethe.1
			}
		}
		set_global_flag = GER_operation_margarethe
	}
}
