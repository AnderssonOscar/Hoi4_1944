industry_extra_start_effects = {
	
	if = {
		limit = {
			date > 1944.1.2
			check_variable = { global.var_industry_extra_start_effects = 0 }
		}

############################### USA RUBBER INDUSTRY ######################################
	USA = {
		modify_building_resources = {
			building = synthetic_refinery
			resource = rubber
			amount = 5
		}
	}

########################################AIR BASE###################################################################################################################

every_state = {
	limit = {

		is_fully_controlled_by = controller
		controller = { num_of_factories > 25}
		NOT = { has_state_flag = war_building_slot_flag }

		AND = {
			air_base  > 1
			air_base  < 6
		}
	}
			
	add_building_construction = {
		type = air_base
		level = 2
		instant_build = yes
	}
}

###########################################################################################################################################################
#Extra for these

every_state = {
	limit = {

		is_core_of = controller
		is_fully_controlled_by = controller
		NOT = { has_state_flag = war_building_slot_flag }

		OR = {
			is_controlled_by = USA
			is_controlled_by = ENG
		}

		free_building_slots = {

			building = arms_factory
			size > 5
			include_locked = yes
		}
	}

	add_extra_state_shared_building_slots = 5
	add_building_construction = {
		type = arms_factory
		level = 5
		instant_build = yes
	}
}


		set_variable = { global.var_industry_extra_start_effects = 1 }
	}
}
