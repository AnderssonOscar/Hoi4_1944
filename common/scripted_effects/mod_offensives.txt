
MOD_SOV_offensives = {
	
	if = {
		limit = {
			tag = SOV
			has_war_with = GER
			has_manpower > 250000
			has_war_support > 0.6
			#alliance_strength_ratio > 1
			surrender_progress < 0.4
			date > 1944.1.1

			set_temp_variable = { days = global.num_days }
			modulo_temp_variable = { days = 120 }
			check_variable = { days = 60 }
		}


			add_timed_idea = {
				idea = major_offensive
				days = 25
			}
	
	}

	if = {
		limit = {
			tag = SOV
			has_war_with = GER
			date > 1944.1.1

			set_temp_variable = { days = global.num_days }
			modulo_temp_variable = { days = 120 }
			check_variable = { days = 83 }

			has_idea = major_offensive
		}

			add_timed_idea = {
				idea = offensive_exhaustion
				days = 20
			}
	
	}
}
