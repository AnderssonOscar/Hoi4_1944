industry_start_effects = {
	
	if = {
		limit = {
			#date > 1944.1.1
			check_variable = { global.var_industry_start_effects = 0 }
		}


####################################################### INDUSTRY #####################################################
every_state = {
	limit = {

		is_core_of = controller
		is_fully_controlled_by = controller

		controller = { num_of_factories > 15}
		NOT = { has_state_flag = war_building_slot_flag }
		NOT = { is_controlled_by = GER } #Germany already got to civs added to states 

		OR = {
			arms_factory  > 1
			industrial_complex > 1
			dockyard > 1
		}
			
		free_building_slots = {
			building = industrial_complex
			size > 1
			include_locked = yes
		}
	}

	add_extra_state_shared_building_slots = 1
	add_building_construction = {
		type = industrial_complex
		level = 1
		instant_build = yes
	}
}


every_state = {
	limit = {

		is_core_of = controller
		is_fully_controlled_by = controller

		controller = { num_of_factories > 15}
		NOT = { has_state_flag = war_building_slot_flag }

		OR = {
			arms_factory  > 1
			industrial_complex > 1
			dockyard > 1
		}
			
		free_building_slots = {

			building = arms_factory
			size > 3
			include_locked = yes
		}
	}

	add_extra_state_shared_building_slots = 3
	add_building_construction = {
		type = arms_factory
		level = 3
		instant_build = yes
	}
}

		set_variable = { global.var_industry_start_effects = 1 }
	}
}
