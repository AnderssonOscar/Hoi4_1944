on_actions = {
	on_startup = {
		effect = {

	#add_named_threat = { threat = 60 name = generic.15.t }
				
	GER = { declare_war_on = { target = ENG type = annex_everything } }

	every_country = {
		limit = { is_in_faction_with = GER }
		add_to_war = {
			targeted_alliance = GER
			enemy = ENG
		}
	}

	ENG = {
		every_country = {
			limit = { is_in_faction_with = ENG }
			add_to_war = {
				targeted_alliance = ENG
				enemy = GER
			}
		}
	}


if = {
	limit = {
		has_dlc = "La Resistance"
	}
		every_state = {
			limit = {
				controller = {has_war = yes}
				NOT = {is_core_of = controller}
				is_fully_controlled_by = controller
			}
			set_resistance = 27
		}
}






GER = {
	add_ideas = {
		tot_economic_mobilisation
		#add_manpower = -3500000
	}
}



###########################################################################################################################################################
#####################################################     MORE INDUSTRY #################################################################################
###########################################################################################################################################################

			every_state = {
				limit = {

					date > 1943.12.29

					is_core_of = controller
					is_fully_controlled_by = controller

					controller = { num_of_factories > 15}
					NOT = { has_state_flag = war_building_slot_flag }

					#controller = { is_on_continent = europe }


					NOT = { is_controlled_by = GER } #Germany already got to civs added to states 
						


					OR = {
						arms_factory  > 1
						industrial_complex > 1
						dockyard > 1
					}
						
					free_building_slots = {
						building = industrial_complex
						size > 1
						include_locked = yes
					}
				}

				add_extra_state_shared_building_slots = 1
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}


			every_state = {
				limit = {

					date > 1943.12.29

					is_core_of = controller
					is_fully_controlled_by = controller

					controller = { num_of_factories > 15}
					NOT = { has_state_flag = war_building_slot_flag }

					#controller = { is_on_continent = europe }


					OR = {
						arms_factory  > 1
						industrial_complex > 1
						dockyard > 1
					}
						

					free_building_slots = {

						building = arms_factory
						size > 3
						include_locked = yes
					}
				}

				add_extra_state_shared_building_slots = 3
				add_building_construction = {
					type = arms_factory
					level = 3
					instant_build = yes
				}
			}





########################################AIR BASE###################################################################################################################


			every_state = {
				limit = {

					date > 1943.12.29


					is_fully_controlled_by = controller

					controller = { num_of_factories > 25}
					NOT = { has_state_flag = war_building_slot_flag }

					#controller = { is_on_continent = europe }

					AND = {
						air_base  > 1
						air_base  < 6
					}
				}
						
				add_building_construction = {
					type = air_base
					level = 2
					instant_build = yes
				}
			}


###########################################################################################################################################################


		#Extra for these


			every_state = {
				limit = {

					date > 1943.12.29

					is_core_of = controller
					is_fully_controlled_by = controller

					NOT = { has_state_flag = war_building_slot_flag }

					OR = {
						is_controlled_by = USA
						is_controlled_by = ENG
					}


					free_building_slots = {

						building = arms_factory
						size > 4
						include_locked = yes
					}
				}

				add_extra_state_shared_building_slots = 4
				add_building_construction = {
					type = arms_factory
					level = 4
					instant_build = yes
				}
			}




###########################################################################################################################################################

			}


		}
