###########################
# 1944 - Downfall: into Leningrad: daily check (docs/CHANGELOG.md section 23)
#
# The game cannot tell where a battle is fought, so this uses the map instead.
# Leningrad (state 195) is the city province 3151 and four around it. On
# 1 January 1944 we hold two of them (6174 and 79, the siege line). The event
# fires once, the first day we also hold 149, between the city and Lake
# Ladoga, and still hold 6174, while the Soviet Union holds the city: the
# city's land neighbours south of the Neva are then all ours, and the next
# step is the city itself. Its fall is announced by the game's own news
# event "The Fall of Leningrad" (news.103).
###########################

on_actions = {
	on_daily_GER = {
		effect = {
			if = {
				limit = {
					NOT = { has_country_flag = GER_leningrad_done }
					has_war_with = SOV
					controls_province = 149
					controls_province = 6174
					SOV = { controls_province = 3151 }
				}
				set_country_flag = GER_leningrad_done
				country_event = { id = ger_leningrad.1 }
			}
		}
	}
}
