
political_actions = {

	TUR_start_embargo_with_axis = {

		icon = generic_prepare_civil_war

		allowed = {
			tag = TUR	
		}

		available = {

			OR = {
				GER = { NOT = { is_embargoed_by = TUR } }
				JAP = { NOT = { is_embargoed_by = TUR } }
				any_country = {
					OR = {
						tag = GER 
						tag = JAP  
					}
					any_allied_country = {
						NOT = { is_embargoed_by = TUR }
					}
				}
			}

			if = {
				limit = {
					any_country = {
						OR = {
							tag = GER
							is_in_faction_with = GER
							is_in_faction_with = JAP
						}
					}
				}
				
				 OR = {
					GER = { surrender_progress > 0.05 }
					NOT = { 766 = { is_fully_controlled_by = THIS } }#south bessarabia
					NOT = { 78 = { is_fully_controlled_by = THIS } }#bessarabia
				}
			}
	}

		visible = {
			date > 1944.1.1
			has_capitulated = no
			NOT = {has_war_with = GER}
			is_ai = yes

			GER = { has_capitulated = no }
			NOT = { is_in_faction_with = GER }

			AND = {
				ENG = {has_capitulated = no}
				USA = {has_capitulated = no}
				SOV = {has_capitulated = no}
			}

			AND = {
				GER = {has_war_with = SOV}
				GER = {has_war_with = ENG}
				GER = {has_war_with = USA}
			}
		}

		fire_only_once = no
		days_re_enable = 14
		cost = 0

		ai_will_do = {
			factor = 1
			modifier = {

				date > 1944.4.15
				factor = 100
			}
		}

	complete_effect = {
		
		set_variable = { global.TUR_start_embargo_with_axis = 1 }

		every_country = {
			limit = { 
				OR = {
					is_in_faction_with = GER
					is_in_faction_with = JAP
				}
				NOT = { is_embargoed_by = TUR } 
			}
			
				country_event = { id = embargo.start.1 }

				if = {
					limit = { has_dlc = "By Blood Alone" } #maybe not needed
						ROOT = { send_embargo = PREV }
					else = {
						ROOT = {add_opinion_modifier = { target = PREV modifier = embargo}}
						ROOT = {add_opinion_modifier = { target = PREV modifier = large_decrease }}
						ROOT = {add_opinion_modifier = { target = PREV modifier = large_decrease }}
					}
				}
		}
		custom_effect_tooltip = embargo_relation_tt
		}
	}


	TUR_stop_embargo_with_axis = {

		icon = generic_prepare_civil_war

		allowed = {
			tag = TUR	
		}

		available = {

		OR = {
			if = {
				limit = {
					any_country = {
						OR = {
							tag = GER
							is_in_faction_with = GER
							is_in_faction_with = JAP
						}
					}
				}

				GER = { surrender_progress < 0.02 }
				766 = { is_fully_controlled_by = THIS } #south bessarabia
				78 = { is_fully_controlled_by = THIS } #bessarabia
			}

				AND = {
					NOT = { GER = { has_war_with = USA } }
					NOT = { GER = { has_war_with = ENG } }
				}
				is_in_faction_with = GER
		}
	}

		visible = {
			date > 1944.4.15
			has_capitulated = no
			is_ai = yes
			check_variable = { global.TUR_start_embargo_with_axis = 1 }
		}

		fire_only_once = no
		cost = 0
		days_re_enable = 14

		ai_will_do = {
			factor = 1
			modifier = {

				date > 1944.1.1
				factor = 100
			}
		}

	complete_effect = {
		set_variable = { global.TUR_start_embargo_with_axis = 0 }

		every_country = {
			limit = { 
				OR = {
					is_in_faction_with = GER
					is_in_faction_with = JAP
				}
				is_embargoed_by = TUR 
			}

				country_event = { id = embargo.stop.1 }

				if = {
					limit = { has_dlc = "By Blood Alone" }
						ROOT = { break_embargo = PREV }
					else = {
						ROOT = {remove_opinion_modifier = { target = PREV modifier = embargo}}
						ROOT = {remove_opinion_modifier = { target = PREV modifier = large_decrease }}
						ROOT = {remove_opinion_modifier = { target = PREV modifier = large_decrease }}
				}
			}
		}

		}
	}
}
