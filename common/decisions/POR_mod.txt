

political_actions = {

	POR_start_embargo_with_everyone = {

		icon = generic_prepare_civil_war

		allowed = {
			tag = POR	
		}

		available = {


			OR = {
				GER = { NOT = { is_embargoed_by = POR } }
				JAP = { NOT = { is_embargoed_by = POR } }
				ENG = { NOT = { is_embargoed_by = POR } }
				USA = { NOT = { is_embargoed_by = POR } }
				any_country = {
					OR = {
						tag = GER 
						tag = JAP  
						tag = ENG
						tag = USA
					}
					any_allied_country = {
						NOT = { is_embargoed_by = POR }
					}
				}
			}



				date > 1944.6.4 

			OR = {
				GER = {has_war_with = ENG}
				GER = {has_war_with = USA}
				GER = {has_war_with = SOV}
			}
		}

		visible = {
			date > 1944.1.1 #1944.6.4 

			has_capitulated = no
			has_war = no

			OR = {
				NOT = { is_in_faction_with = GER }
				NOT = { is_in_faction_with = USA }
				NOT = { is_in_faction_with = SOV }
				NOT = { is_in_faction_with = JAP }
			}

			#check_variable = { global.POR_start_embargo_with_axis = 0 }
		} 

		fire_only_once = no
		days_re_enable = 14
		cost = 0

		ai_will_do = { 
			factor = 1
			modifier = {
				date > 1944.1.1
				factor = 100
			}
		}

	complete_effect = {
		set_variable = { global.POR_start_embargo_with_axis = 1 }

		every_country = {
			limit = { 
				OR = {
					is_in_faction_with = GER
					is_in_faction_with = JAP
					is_in_faction_with = USA
					is_in_faction_with = ENG
					is_in_faction_with = SOV
				}
			}

				if = {
					limit = { has_dlc = "By Blood Alone" }
						ROOT = { send_embargo = PREV }
					else = {
						ROOT = {add_opinion_modifier = { target = PREV modifier = embargo}}
						ROOT = {add_opinion_modifier = { target = PREV modifier = large_decrease }}
						ROOT = {add_opinion_modifier = { target = PREV modifier = large_decrease }}
					}
				}
		}
		custom_effect_tooltip = embargo_relation_tt
		}
	}


	POR_stop_embargo_with_everyone = {

		icon = generic_prepare_civil_war

		allowed = {
			tag = POR	
		}

		available = {

			OR = {
				AND = {
					NOT = { GER = {has_war_with = ENG} }
					NOT = { GER = {has_war_with = USA} }
					NOT = { GER = {has_war_with = SOV} }
				}

				has_war = yes
				is_in_faction_with = GER
				is_in_faction_with = USA
				is_in_faction_with = SOV
			}
		}

		visible = {
			date > 1944.1.1
			has_capitulated = no

			check_variable = { global.POR_start_embargo_with_axis = 1 }
		}

		fire_only_once = no
		cost = 0
		days_re_enable = 14

		ai_will_do = {
			factor = 1
			modifier = {

				date > 1944.1.1
				factor = 100
			}
		}

	complete_effect = {
		set_variable = { global.POR_start_embargo_with_axis = 0 }

		every_country = {
			limit = { 
				OR = {
					is_in_faction_with = GER
					is_in_faction_with = JAP
					is_in_faction_with = USA
					is_in_faction_with = ENG
					is_in_faction_with = SOV
				} 
			}

				if = {
					limit = { has_dlc = "By Blood Alone" }
						ROOT = { break_embargo = PREV }
					else = {
						ROOT = {remove_opinion_modifier = { target = PREV modifier = embargo}}
						ROOT = {remove_opinion_modifier = { target = PREV modifier = large_decrease }}
						ROOT = {remove_opinion_modifier = { target = PREV modifier = large_decrease }}
				}
			}
		}
		
		}
	}
}
