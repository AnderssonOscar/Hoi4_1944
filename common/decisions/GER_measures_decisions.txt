collaborationist_formation = {

	# Expand the KONR: from 27 February 1945 (Oscar's date for the plans to
	# expand it), and only after the author's "Recruit Andrey Vlasov" (which
	# creates the "Russische Befreiungsarmee" template). The author's own
	# "Russian Liberation Army" decision raises the 600th, 650th and 700th
	# divisions; this one adds a 4th and 5th, or manpower (event ger_measures.1).
	GER_measures_konr_expand = {
		icon = generic_army_support
		allowed = { original_tag = GER }
		visible = {
			date > 1945.2.26
			has_war_with = SOV
			has_capitulated = no
			has_template = "Russische Befreiungsarmee"
		}
		available = {
			has_war_with = SOV
			has_template = "Russische Befreiungsarmee"
		}
		cost = 50
		fire_only_once = yes
		ai_will_do = { factor = 1 }
		complete_effect = {
			custom_effect_tooltip = GER_measures_konr_expand_tt
			hidden_effect = { country_event = { id = ger_measures.1 } }
		}
	}
}
