
political_actions = {

	SPA_start_embargo_with_axis = {

		icon = generic_prepare_civil_war

		allowed = {
			tag = SPA	
		}

		available = {


			OR = {
				GER = { NOT = { is_embargoed_by = SPA } }
				JAP = { NOT = { is_embargoed_by = SPA } }
				any_country = {
					OR = {
						tag = GER 
						tag = JAP  
					}
					any_allied_country = {
						NOT = { is_embargoed_by = SPA }
					}
				}
			}

			if = {
				limit = {
					any_country = {
						OR = {
							tag = GER
							is_in_faction_with = GER
							is_in_faction_with = JAP
						}
					}
				}

				#SPA = { is_embargoing = GER }
				
				NOT = { 16 = { is_fully_controlled_by = THIS } } #Paris
				
				 OR = {
					NOT = { 31 = { is_fully_controlled_by = THIS } }#midi pyrenees
					NOT = { 806 = { is_fully_controlled_by = THIS } }#pyrenees-atlant
				}
		}
	}

		visible = {
			date > 1944.1.1
			has_capitulated = no
			NOT = {has_war_with = GER}
			is_ai = yes
			GER = { has_capitulated = no }
			NOT = { is_in_faction_with = GER }

			#check_variable = { global.SPA_start_embargo_with_axis = 0 }

			AND = {
				ENG = {has_capitulated = no}
				USA = {has_capitulated = no}
				SOV = {has_capitulated = no}
			}

			AND = {
				GER = {has_war_with = SOV}
				GER = {has_war_with = ENG}
				GER = {has_war_with = USA}
			}
		}

		fire_only_once = no
		days_re_enable = 14
		cost = 0

		ai_will_do = { #must lose paris and pyrenees
			factor = 1
			modifier = {

				date > 1944.1.1
				factor = 100
			}
		}

	complete_effect = {
		
		set_variable = { global.SPA_start_embargo_with_axis = 1 }

		every_country = {
			limit = { 
				OR = {
					is_in_faction_with = GER
					is_in_faction_with = JAP
				}
				NOT = { is_embargoed_by = SPA }
			}

				country_event = { id = embargo.start.1 }

				if = {
					limit = { has_dlc = "By Blood Alone" }
						ROOT = { send_embargo = PREV }
					else = {
						ROOT = {add_opinion_modifier = { target = PREV modifier = embargo}}
						ROOT = {add_opinion_modifier = { target = PREV modifier = large_decrease }}
						ROOT = {add_opinion_modifier = { target = PREV modifier = large_decrease }}
					}
				}
		}
		custom_effect_tooltip = embargo_relation_tt
		}
	}


	SPA_stop_embargo_with_axis = {

		icon = generic_prepare_civil_war

		allowed = {
			tag = SPA	
		}

		available = {

		OR = {
			if = {
				limit = {
					any_country = {
						OR = {
							tag = GER
							is_in_faction_with = GER
							is_in_faction_with = JAP
						}
					}
				}
				
				16 = { is_fully_controlled_by = THIS } #Paris
				31 = { is_fully_controlled_by = THIS } #midi pyrenees
				806 = { is_fully_controlled_by = THIS } #pyrenees-atlant
			}

				AND = {
					NOT = { GER = { has_war_with = USA } }
					NOT = { GER = { has_war_with = ENG } }
				}
				is_in_faction_with = GER
		}
	}

		visible = {
			date > 1944.1.1
			has_capitulated = no
			is_ai = yes
			check_variable = { global.SPA_start_embargo_with_axis = 1 }
		}

		fire_only_once = no
		cost = 0
		days_re_enable = 14

		ai_will_do = { #must lose paris and pyrenees
			factor = 1
			modifier = {

				date > 1944.1.1
				factor = 100
			}
		}

	complete_effect = {
		set_variable = { global.SPA_start_embargo_with_axis = 0 }

		every_country = {
			limit = { 
				OR = {
					is_in_faction_with = GER
					is_in_faction_with = JAP
				} 
				is_embargoed_by = SPA
			}

				country_event = { id = embargo.stop.1 }

				if = {
					limit = { has_dlc = "By Blood Alone" }
						ROOT = { break_embargo = PREV }
					else = {
						ROOT = {remove_opinion_modifier = { target = PREV modifier = embargo}}
						ROOT = {remove_opinion_modifier = { target = PREV modifier = large_decrease }}
						ROOT = {remove_opinion_modifier = { target = PREV modifier = large_decrease }}
				}
			}
		}

		}
	}
}
