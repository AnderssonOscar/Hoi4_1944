			}

			date > 2000.1.1
		}

		available = {
			surrender_progress > 0.1
			has_war_support > 0.5
		}

		fire_only_once = yes
		days_remove = 7
		cost = 50

		remove_trigger = {
			OR = {
					has_war = no
					not = { has_idea = tot_economic_mobilisation }		
			}
		}

		ai_will_do = {
			base = 1
			modifier = {
				add = 10
				has_equipment = {
					anti_air_equipment < 1
				}
			}
		}
  
		remove_effect = {

			every_controlled_state = {
				limit = {
					anti_air_building > 0
					is_fully_controlled_by = controller
			}

			set_temp_variable = { anti_air_eq = 0 }
			add_to_temp_variable = { var = anti_air_eq value = 1 }
			multiply_temp_variable = { anti_air_eq = 50 }

			ROOT = {
					add_equipment_to_stockpile = {
					type = anti_air_equipment_2
					amount = anti_air_eq
				}
			}
			remove_building = { type = anti_air_building level = 1 }

			}
			add_war_support = -0.01
		}
	}
