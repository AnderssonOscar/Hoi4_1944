ROM_hold_soviet_offensive = {

	allowed = { tag = ROM }

	enable = {
		OR = {
			SOV = { has_idea = major_offensive } 

			AND = {
				SOV = { NOT = { has_idea = major_offensive } }
				SOV = { NOT = { has_idea = offensive_exhaustion } } 
			}
		}
		is_historical_focus_on = yes

		has_war_with = SOV 
		NOT = { has_war_with = BUL }
		NOT = { has_war_with = GER }
		is_in_faction_with = GER

		date > 1944.1.1
	#	date < 1944.9.1
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		country_trigger = { tag = SOV }
		ordertype = front
		execute_order = yes
		execution_type = careful
	}
}
