# 3. Clausewitz: Seelow has fallen or the enemy is next to Berlin
country_event = {
	id = festung_berlin.3
	title = festung_berlin.3.t
	desc = festung_berlin.3.d
	picture = GFX_report_event_german_reichstag_bombed

	is_triggered_only = yes

	option = {
		name = festung_berlin.3.a
		custom_effect_tooltip = festung_berlin.3.a.tt
		hidden_effect = {
			GER_festung_berlin_emergency_units = yes
			64 = { add_dynamic_modifier = { modifier = GER_battle_for_berlin } }
			if = {
				limit = {
					is_ai = no
					NOT = { has_completed_focus = GER_festung_cities }
					controls_province = 6521
				}
				64 = { add_province_modifier = { static_modifiers = { GER_festung_berlin_city_modifier } province = { id = 6521 } } }
				set_country_flag = GER_festung_berlin_city_added
			}
			every_country = {
				limit = { OR = { tag = ROOT has_war_with = ROOT } }
				news_event = { id = festung_berlin.20 days = 1 }
			}
			country_event = { id = festung_berlin.4 days = 3 }
		}
	}
}

# 4. Weidling takes command (3 days after Clausewitz, if Berlin still holds)
country_event = {
	id = festung_berlin.4
	title = festung_berlin.4.t
	desc = festung_berlin.4.d
	picture = GFX_report_event_german_inspect_troops

	is_triggered_only = yes

	trigger = {
		controls_province = 6521
	}

	option = {
		name = festung_berlin.4.a
		if = {
			limit = {
				has_character = GER_helmuth_weidling
				GER_helmuth_weidling = { NOT = { has_trait = urban_assault_specialist } }
			}
			GER_helmuth_weidling = { add_unit_leader_trait = urban_assault_specialist }
		}
	}
}
