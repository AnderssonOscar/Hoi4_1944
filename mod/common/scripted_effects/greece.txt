GRE_setup_resistance = {

	GRE = {
		if = {
			limit = {
				date > 1943.12.30
				is_subject_of = GER
				NOT = { has_global_flag = GRE_setup_resistance }
			}
				set_global_flag = GRE_setup_resistance

			every_state = {
				limit = {
					is_core_of = THIS
					controller = {
						has_government = fascism
						tag = GRE
					}
				}

				force_enable_resistance = {
					occupier = THIS.controller
					occupied = THIS
				}

				start_resistance = THIS
				set_resistance = 27
				set_compliance = 38
				add_resistance_target = 34
			}
		}
	}
