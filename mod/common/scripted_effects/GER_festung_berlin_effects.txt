# 1944 - Downfall: Festung Berlin (see docs/CHANGELOG.md section 9)
#
# Fort top-ups and emergency units for events/GER_festung_berlin_events.txt.
# Country scope: ROOT is Germany.
#
# Forts are only ever topped up to a target level, never added on top: each
# province checks its current bunker level once and adds exactly the
# difference. Forts the author's Festung Cities focus or Oder-Neisse Defence
# decision already built are therefore not stacked on, and nothing goes above
# the target. Only provinces Germany still controls are built.
#   Berlin (6521)                               -> 2, 4, 5 (steps 1-3)
#   around Berlin (375 3499 9428 11444 11505)   -> 1, 2    (steps 1-2)
#   Seelow (9496)                               -> 2, 4    (Seelow steps 1-2)
# 5 in Berlin is the author's own Festung level. AI armies in 'careful' mode
# do not attack provinces with fort level 5 or more, so nothing here goes higher.
###########################

# Berlin Defence Area, step 1: Berlin to 2, the provinces around it to 1.
GER_festung_berlin_forts_step_1 = {
	64 = {

			limit = { ROOT = { controls_province = 6521 } }

				limit = { any_province_building_level = { province = { id = 6521 } building = bunker level < 1 } }
