###########################
# 1944 - Downfall: the home front, 1944-45: effects (docs/CHANGELOG.md section 14)
###########################

# The effects of the events. They live here so that the in-game test runs
# exactly the same code as the events.

# 1. 15 August 1944: women's labour service extended from 45 to 50. The men
# they replace in industry are released: a one-time 10,000 manpower.
GER_homefront_female_labour_to_50 = {
	add_manpower = 10000
}

# 2. 14 October 1944: Rommel's forced suicide. He leaves German service,
# unless the author's July plot event already retired him.
GER_homefront_rommel_dies = {
	if = {
		limit = { has_character = GER_erwin_rommel }
		retire_character = GER_erwin_rommel
	}
}

# 3. 17 November 1944: RAD women to the Flak. The soldiers they replace are
# released (7,500), and a small lasting State AA bonus.
GER_homefront_flak_auxiliaries = {
	add_manpower = 7500
	add_ideas = GER_homefront_flak_auxiliaries
}

# 4. 19 December 1944: an air force for Vlasov's army. The political price,
# its men (about 5,000) and a small formation of the types it flew: Bf 109 G
# fighters and Ju 87 dive bombers. The design is named (variant_name): given
# by the general type, the game handed out Ta 152 fighters and pre-war
# ground-attack planes (CHANGELOG section 17).
GER_homefront_vlasov_air_force = {
	add_political_power = -25
	add_manpower = 5000
	add_equipment_to_stockpile = { type = small_plane_airframe_2 amount = 20 producer = GER variant_name = "Bf 109 G" }
	add_equipment_to_stockpile = { type = small_plane_cas_airframe_1 amount = 10 producer = GER variant_name = "Ju 87" }
}

# 5. 6 January 1945: the Volksopfer collection for the Volkssturm. Old,
# mixed kit rather than new weapons.
GER_homefront_volksopfer = {
	add_political_power = -25
	add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 4000 }
	add_equipment_to_stockpile = { type = support_equipment_1 amount = 1000 }
}

# 6. 30 January 1945: the Wilhelm Gustloff. About 1,600 naval personnel,
# U-boat trainees, auxiliaries and wounded soldiers were aboard; nearly all
# of them died.
GER_homefront_gustloff = {
	add_manpower = -1500
}

# 7. 12 February 1945: women and girls called to auxiliary service with the
# Volkssturm. Men released from support duties, at a small cost in stability.
GER_homefront_volkssturm_women = {
	add_manpower = 5000
	add_stability = -0.02
}

# 8. 5 March 1945: the class of 1929 called up, with shortened training.
GER_homefront_class_of_1929 = {
	add_ideas = GER_homefront_class_of_1929
}
