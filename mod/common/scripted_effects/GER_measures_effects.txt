# The effects of the decisions and events. They live here so that the in-game
# test runs exactly the same code as the decisions and events.

# KONR, option A: two more divisions of the author's Russian Liberation Army
# template, at low experience and poorly armed: 30% of their equipment, with
# old or captured rifles. The game gives a division only one rifle model, so
# each division rolls it: German rifles from old depot stocks 50%, captured
# Soviet 40%, captured Italian 5%, captured French 5%.
# The equipment they arrive with is paid from our stockpile: a full division
# needs 1,510 rifles, 30 support equipment and 12 guns (15 infantry battalions,
# an engineer and an artillery company; the template lists 16 battalions, but
# two share the slot x = 1, y = 2, so the game builds 15), so 30% is 453, 9 and
# 4 (rounded up). Measured in the game: 453 rifles per division.
# The 4th division is raised at once, the 5th after 60 days (event
# ger_measures.2), each paid when it is raised.
#
# The payment is taken type by type (oldest first): removing by the general
# type (infantry_equipment) would take the amount from every type in stock
# (measured in the game, CHANGELOG section 17).
# Raised in Wuerttemberg (52) or Baden (50), where the KONR divisions were
# formed (Muensingen, Heuberg), else in any other German core state we hold.
GER_measures_konr_divisions = {
	GER_measures_konr_pay = yes
	GER_measures_konr_4th_division = yes
	hidden_effect = { country_event = { id = ger_measures.2 days = 60 } }
}

GER_measures_konr_4th_division = {
	random_owned_controlled_state = {
		limit = { is_core_of = ROOT }
		prioritize = { 52 50 }
		random_list = {
			seed = random
			50 = { # German rifles from old depot stocks (Gewehr 98)
				create_unit = {
					division = "name = \"4th KONR Infantry Division\" division_template = \"Russische Befreiungsarmee\" start_experience_factor = 0.1 start_equipment_factor = 0.3 force_equipment_variants = { infantry_equipment_0 = { owner = \"GER\" } }"
					owner = ROOT
				}
			}
			40 = { # captured Soviet rifles (Mosin-Nagant)
				modifier = { factor = 0 NOT = { country_exists = SOV } }
